const { Telegraf } = require('telegraf'); // Токен бота и токен платёжного провайдера из @BotFather const bot = new Telegraf('ВАШ_BOT_TOKEN'); const PROVIDER_TOKEN = 'ВАШ_PROVIDER_TOKEN'; // Команда для выставления счёта на оплату bot.command('pay', (ctx) => { return ctx.replyWithInvoice({ title: 'Оплата заказа', description: 'Оплата услуг через официальный шлюз Telegram', payload: 'order_12345', // Внутренний ID заказа для вашей базы provider_token: PROVIDER_TOKEN, currency: 'RUB', // Или UAH, USD, EUR в зависимости от провайдера prices: [ { label: 'Стоимость услуги', amount: 50000 } // Сумма в копейках/центах (500.00) ], need_phone_number: true, // Запросить телефон для чека при необходимости send_phone_number_to_provider: true }); }); // Обязательное подтверждение перед списанием средств bot.on('pre_checkout_query', (ctx) => { return ctx.answerPreCheckoutQuery(true); }); // Обработка успешного платежа bot.on('successful_payment', (ctx) => { const payment = ctx.message.successful_payment; console.log('Оплата прошла успешно:', payment); ctx.reply(`Оплата на сумму ${payment.total_amount / 100} ${payment.currency} успешно принята!`); }); bot.launch();
